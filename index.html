Kết Nối Google Sheets Tức Thời (Live Sync)

Tự động lưu bài làm của học sinh vào Google Sheet ngay khi nộp, hiển thị trực tiếp và xóa dữ liệu đồng bộ.

Hướng dẫn 3 bước kết nối Google Sheets (Bấm vào để xem mã Apps Script)
  1. Tạo một file Google Sheets mới trên Google Drive của giáo viên.
  2. Vào menu Tiện ích mở rộng (Extensions) > Apps Script, xóa toàn bộ code cũ và dán đoạn mã bên dưới vào:
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  if (data.action === "clear_all") {
    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).clearContent();
    }
    return ContentService.createTextOutput(JSON.stringify({status: "success", msg: "Cleared"}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Thời gian", "Họ và tên", "Lớp", "Đề thi", "Tổng điểm", "Phần I", "Phần II", "Phần III", "Thời gian làm bài"]);
  }
  
  var d = data.data;
  sheet.appendRow([d.timestamp, d.student_name, d.student_class, d.exam_id, d.total_score, d.p1_score, d.p2_score, d.p3_score, d.duration]);
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = [];
  if (rows.length > 1) {
    for (var i = 1; i < rows.length; i++) {
      result.push({
        timestamp: rows[i][0],
        student_name: rows[i][1],
        student_class: rows[i][2],
        exam_id: rows[i][3],
        total_score: rows[i][4],
        p1_score: rows[i][5],
        p2_score: rows[i][6],
        p3_score: rows[i][7],
        duration: rows[i][8]
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({status: "success", data: result}))
    .setMimeType(ContentService.MimeType.JSON);
}
